cmake_minimum_required(VERSION 3.16)
project(uran_framework VERSION 2.0.0 LANGUAGES CXX)
include(CTest)
set(native_default OFF)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(native_default ON)
endif()
option(URAN_BUILD_NATIVE "Build the Linux native host, loader and file backend" ${native_default})
option(URAN_BUILD_ROS2 "Build the optional ROS 2 host and pluginlib backend" OFF)
option(URAN_FETCH_DEPENDENCIES "Download pinned nlohmann_json if unavailable" ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(cmake/Dependencies.cmake)
add_subdirectory(src/uran_plugin_api)
add_subdirectory(src/uran_core_framework)
if(URAN_BUILD_NATIVE)
  add_subdirectory(src/uran_platform_native)
endif()
add_subdirectory(src/uran_core_example_plugins)
if(URAN_BUILD_ROS2)
  if(NOT URAN_BUILD_NATIVE)
    message(FATAL_ERROR "The supplied ROS 2 host uses the native scheduler; enable URAN_BUILD_NATIVE")
  endif()
  add_subdirectory(src/uran_ros2_adapter)
  add_subdirectory(src/uran_bringup)
endif()
install(DIRECTORY config/ DESTINATION share/uran_framework/config)
if(URAN_BUILD_NATIVE)
  install(PROGRAMS scripts/uran_plugin.py DESTINATION bin RENAME uran-plugin)
endif()
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
