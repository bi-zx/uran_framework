cmake_minimum_required(VERSION 3.16)
project(uran_framework VERSION 2.1.0 LANGUAGES CXX)
include(CTest)
set(native_default OFF)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(native_default ON)
endif()
option(URAN_BUILD_NATIVE "Build the Linux native host, loader and file backend" ${native_default})
option(URAN_FETCH_DEPENDENCIES "Download pinned nlohmann_json if unavailable" ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(cmake/Dependencies.cmake)
add_subdirectory(src/uran_plugin_api)
add_subdirectory(src/uran_core_framework)
if(URAN_BUILD_NATIVE)
  add_subdirectory(src/uran_platform_native)
  add_subdirectory(src/uran_worker_adapter)
endif()
add_subdirectory(src/uran_core_example_plugins)
install(DIRECTORY config/ DESTINATION share/uran_framework/config)
install(DIRECTORY schemas/ DESTINATION share/uran_framework/schemas)
install(DIRECTORY python/uran_worker_sdk/ DESTINATION share/uran_framework/python/uran_worker_sdk
  PATTERN "build" EXCLUDE PATTERN "*.egg-info" EXCLUDE
  PATTERN "__pycache__" EXCLUDE PATTERN "*.pyc" EXCLUDE)
install(DIRECTORY python/uran_deploy/ DESTINATION share/uran_framework/python/uran_deploy
  PATTERN "build" EXCLUDE PATTERN "*.egg-info" EXCLUDE
  PATTERN "__pycache__" EXCLUDE PATTERN "*.pyc" EXCLUDE)
if(URAN_BUILD_NATIVE)
  install(PROGRAMS scripts/uran_plugin.py DESTINATION bin RENAME uran-plugin)
  install(PROGRAMS scripts/uran_deploy.py DESTINATION bin RENAME uran-deploy)
endif()
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
