cmake_minimum_required(VERSION 3.16)
project(uran_platform_native VERSION 2.1.0 LANGUAGES CXX)
include(../../cmake/Package.cmake)
set(URAN_FILESYSTEM_LIBS)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND
   CMAKE_CXX_COMPILER_VERSION VERSION_LESS 8)
  list(APPEND URAN_FILESYSTEM_LIBS stdc++fs)
endif()
if(NOT TARGET uran::uran_core_framework)
  find_package(uran_core_framework 2 REQUIRED CONFIG)
endif()
add_library(${PROJECT_NAME} SHARED src/native_platform.cpp src/native_plugin_loader.cpp src/file_state_storage.cpp)
uran_library(${PROJECT_NAME} "find_dependency(uran_core_framework 2)")
target_link_libraries(${PROJECT_NAME} PUBLIC uran::uran_core_framework PRIVATE ${CMAKE_DL_LIBS} ${URAN_FILESYSTEM_LIBS})
add_executable(uran_runtime src/main.cpp)
target_link_libraries(uran_runtime PRIVATE ${PROJECT_NAME} ${URAN_FILESYSTEM_LIBS})
set_target_properties(uran_runtime PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
install(TARGETS uran_runtime RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
