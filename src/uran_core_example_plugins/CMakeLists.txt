cmake_minimum_required(VERSION 3.16)
project(uran_core_example_plugins VERSION 2.1.0 LANGUAGES CXX)
include(../../cmake/Package.cmake)
if(NOT TARGET uran::uran_plugin_api)
  find_package(uran_plugin_api 2 REQUIRED CONFIG)
endif()
add_library(${PROJECT_NAME} SHARED src/loopback_protocol_plugin.cpp)
uran_library(${PROJECT_NAME} "find_dependency(uran_plugin_api 2)")
target_link_libraries(${PROJECT_NAME} PUBLIC uran::uran_plugin_api)
set_target_properties(${PROJECT_NAME} PROPERTIES INSTALL_RPATH "$ORIGIN")
configure_file(plugin.json.in plugin.json @ONLY)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/plugin.json" DESTINATION share/uran_framework/plugins/loopback)
