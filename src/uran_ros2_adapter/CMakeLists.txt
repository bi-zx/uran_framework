cmake_minimum_required(VERSION 3.16)
project(uran_ros2_adapter VERSION 2.0.0 LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
if(NOT TARGET uran::uran_platform_native)
  find_package(uran_platform_native 2 REQUIRED CONFIG)
endif()
if(NOT TARGET uran::uran_core_example_plugins)
  find_package(uran_core_example_plugins 2 REQUIRED CONFIG)
endif()
add_library(uran_ros2_plugin_exports SHARED src/plugin_exports.cpp)
target_link_libraries(uran_ros2_plugin_exports uran::uran_core_example_plugins)
ament_target_dependencies(uran_ros2_plugin_exports pluginlib)
pluginlib_export_plugin_description_file(uran_ros2_adapter plugin_description.xml)
add_executable(uran_runtime_node src/main.cpp)
target_compile_features(uran_runtime_node PRIVATE cxx_std_17)
target_include_directories(uran_runtime_node PRIVATE include)
target_link_libraries(uran_runtime_node uran::uran_platform_native)
ament_target_dependencies(uran_runtime_node pluginlib rclcpp std_msgs std_srvs)
set_target_properties(uran_runtime_node PROPERTIES INSTALL_RPATH "$ORIGIN/..")
set_target_properties(uran_ros2_plugin_exports PROPERTIES INSTALL_RPATH "$ORIGIN")
install(TARGETS uran_runtime_node DESTINATION lib/${PROJECT_NAME})
install(TARGETS uran_ros2_plugin_exports LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_include_directories(include)
ament_export_dependencies(pluginlib)
ament_package()
